```sh
const hsl=(h)=>{const r=parseInt(h.slice(1,3),16)/255,g=parseInt(h.slice(3,5),16)/255,b=parseInt(h.slice(5,7),16)/255;const mx=Math.max(r,g,b),mn=Math.min(r,g,b);const l=(mx+mn)/2,d=mx-mn;let hh=0,s=0;if(d){s=d/(1-Math.abs(2*l-1));hh=mx===r?60*(((g-b)/d)%6):mx===g?60*(2+(b-r)/d):60*(4+(r-g)/d);}if(hh<0)hh+=360;return [hh,s,l];};
const hex=(hh,s,l)=>{const cc=(1-Math.abs(2*l-1))*s,x=cc*(1-Math.abs(hh/60%2-1));let r,g,b;if(hh<60){[r,g,b]=[cc,x,0];}else if(hh<120){[r,g,b]=[x,cc,0];}else if(hh<180){[r,g,b]=[0,cc,x];}else if(hh<240){[r,g,b]=[0,x,cc];}else if(hh<300){[r,g,b]=[x,0,cc];}else{[r,g,b]=[cc,0,x];}const m=l-cc/2;const f=(v)=>Math.round((v+m)*255).toString(16).padStart(2,"0");return "#"+f(r)+f(g)+f(b);};
```

Крытэр 3 — гард падае на пустым `use` новага токена і на выдаленым запісе
правоў шрыфта, зелёны пасля аднаўлення. Кожны эксперымент пачынаецца і
сканчаецца аднаўленнем канона з git (`git checkout`), таму пасля дэмы
канон застаецца ў камітнутым стане. Вывад падзенняў прапушчаны праз
`sort`: AssertionError ідзе па stderr, лічбы — па stdout, і іх узаемны
парадак на пайпе не вызначаны (rule 11 — захвачаны выход павінен быць
дэтэрмінаваным).

```sh
git checkout -q -- docs/design/visual-language.md
node --input-type=module -e 'import {readFileSync,writeFileSync} from "node:fs"; const p="docs/design/visual-language.md"; const t=readFileSync(p,"utf8"); const nt=t.replace(/("texture\.paper-grain".*?"use": )"[^"]*"/, "$1\"\""); if (nt===t) { console.error("no mutation"); process.exit(1); } writeFileSync(p, nt);'
node --test test/design-tokens.test.mjs 2>&1 | grep -E "AssertionError|^ℹ (pass|fail)" | sort
```
