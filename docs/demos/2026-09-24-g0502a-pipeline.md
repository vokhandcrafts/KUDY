```python
import subprocess
js = """
import { acceptFix, defaultPipelineConfig } from './core/pipeline/pipeline.ts';
import { initialPipelineState } from './core/pipeline/types.ts';

const cfg = defaultPipelineConfig; // dwellMs: 6000 — from services/config, not a literal
const deg = (m) => m / 111195;
const T = (s) => 1_000_000 + s * 1000;
const base = { lat: 54.35, lng: 18.65 };
const north = (p, m) => ({ lat: p.lat + deg(m), lng: p.lng });
const crane = new Map([['stop-crane', { ...north(base, 20), radius: 50 }]]);

let state = initialPipelineState;
const step = (name, fix, candidates = crane) => {
  const r = acceptFix(state, fix, candidates, Number.MAX_SAFE_INTEGER, cfg);
  if (r.accepted) {
    const dwellEvents = r.events.filter((e) => e.type === 'DwellCompleted');
    const tail = dwellEvents.map((e) => ' | DwellCompleted(' + e.stopId + ', radius ' + e.radius + ')').join('');
    console.log(name + '-> accepted; smoothed ' + r.fix.lat.toFixed(5) + ',' + r.fix.lng.toFixed(5) +
      ' | window ' + r.state.window.length + (dwellEvents.length ? tail : ' | dwell events: 0'));
    state = r.state;
  } else {
    console.log(name + '-> rejected: ' + r.reason);
  }
};

step('corrupt   NaN lat        ', { lat: Number.NaN, lng: 18.65, accuracy: 5, at: T(0) });
step('accuracy  60 m claim    ', { ...base, accuracy: 60, at: T(1) });
step('walk      at the quay   ', { ...base, accuracy: 5, at: T(1) });
step('bus       100 m in 1 s  ', { ...north(base, 100), accuracy: 5, at: T(2) });
step('walk      +10 m in 4 s  ', { ...north(base, 10), accuracy: 5, at: T(5) });
step('walk      +20 m in 4 s  ', { ...north(base, 20), accuracy: 5, at: T(9) });
step('stand     +0 m in 4 s   ', { ...north(base, 20), accuracy: 5, at: T(13) });
"""
subprocess.run(['node', '--no-warnings', '--experimental-strip-types', '--input-type=module', '-e', js], check=True)
```
