Executable proof for issue #190 (`services/download`). The grant client is the download side of `/v1/grant` (`09` §5): the request is exactly `{route_id, version, locale, tier, paths[]}` with a Bearer credential from an injected port, paths come only from the verified lock of the layer, every code of the closed list of `19` §3.6 has its own outcome, 503 retries honour `Retry-After`, and an unknown answer fails closed. The transport, credential, clock and byte source are ports — no network in the core; no secret, token or signed URL is printed here.

The request shape, the Bearer port, the 503 retry honouring `Retry-After`, and one outcome per closed code — an answer outside the list is a named unknown failure (criteria 1, 2):

```sh
node --experimental-strip-types -e 'const { requestGrant } = await import("./services/download/grant.ts");
const key = { routeId: "route-x", version: "1", locale: "be", tier: "extended" };
const lock = [{ path: "stops.json", bytes: 13, sha256: "a".repeat(64) }];
let grantCalls = 0;
const requests = [];
const transport = async (req) => {
  grantCalls += 1;
  requests.push(req);
  if (grantCalls === 1) return { status: 503, headers: { "retry-after": "7" }, body: { error: { code: "entitlement_unavailable" } } };
  return { status: 200, headers: {}, body: { lock_url: "https://cdn.test/lock.json", urls: req.body.paths.map((p) => ({ path: p, url: "/private/token-" + p, expires_at: 600000 })) } };
};
const delays = [];
const deps = { transport, credential: async () => "device-secret (redacted)", delay: async (ms) => delays.push(ms) };
const granted = await requestGrant({ ...key, lock }, deps);
console.log("wire body of the first POST /v1/grant:");
console.log("  " + JSON.stringify(requests[0].body));
console.log("bearer credential   : from the injected port (redacted)");
console.log("503 Retry-After: 7  : delay " + delays[0] + " ms, then " + granted.kind);
console.log("granted urls        : " + granted.urls.urls.length + " for " + granted.urls.urls.map((u) => u.path).join(", "));
const deny = async (status, code) => {
  const outcome = await requestGrant({ ...key, lock }, { ...deps, transport: async () => ({ status, headers: {}, body: { error: { code } } }) });
  return outcome.kind;
};
console.log("403 no_entitlement        -> " + (await deny(403, "no_entitlement")));
console.log("403 device_auth_failed    -> " + (await deny(403, "device_auth_failed")));
console.log("400 invalid_request       -> " + (await deny(400, "invalid_request")));
console.log("418 teapot (outside list) -> " + (await deny(418, "teapot")));
' 2>/dev/null
```

A signed URL expiring mid-download: the source re-grants the remaining portion and continues, and the already-verified file is never fetched again (criterion 3):

```sh
node --experimental-strip-types -e 'const fs = await import("node:fs");
const { activate, parseLock } = await import("./services/download/download.ts");
const { createGrantFetchSource } = await import("./services/download/grant.ts");
const { createNodeDownloadStore, nodeSha256 } = await import("./services/download/nodeDownloadStore.ts");
const { openDatabase } = await import("./services/db/db.ts");
const { nodeSqliteDriver } = await import("./services/db/test-fixture.ts");
const root = "/tmp/g0402b-demo-2";
fs.rmSync(root, { recursive: true, force: true });
const enc = (s) => new TextEncoder().encode(s);
const stops = enc("{\"stops\":[]}\n");
const audio = enc("audio-bytes-0123456789abcdef");
const lock = [
  { path: "stops.json", bytes: stops.length, sha256: await nodeSha256(stops) },
  { path: "audio/story-1.m4a", bytes: audio.length, sha256: await nodeSha256(audio) },
];
const sources = { "stops.json": stops, "audio/story-1.m4a": audio };
const key = { routeId: "route-x", version: "1", locale: "be", tier: "extended" };
let grants = 0;
const grantPaths = [];
const transport = async (req) => {
  grants += 1;
  grantPaths.push([...req.body.paths]);
  return { status: 200, headers: {}, body: { lock_url: "https://cdn.test/lock.json", urls: req.body.paths.map((p) => ({ path: p, url: "/private/" + p, expires_at: 600000 })) } };
};
let expired = true;
const byteFetches = [];
const fetchBytes = async (url) => {
  byteFetches.push(url);
  const p = url.slice("/private/".length);
  if (p === "audio/story-1.m4a" && expired) {
    expired = false;
    return { status: 403, body: enc(JSON.stringify({ error: { code: "url_expired" } })) };
  }
  return { status: 200, body: sources[p] };
};
const fetch = createGrantFetchSource(
  { key, entries: parseLock(lock).entries },
  { transport, credential: async () => "redacted", fetchBytes, delay: async () => {}, now: () => 0 },
);
const driver = nodeSqliteDriver();
openDatabase(driver);
const result = await activate({ ...key, lock }, { store: createNodeDownloadStore(root), fetch, sha256: nodeSha256, driver });
console.log("activation           : " + result.status);
console.log("grant rounds         : " + grants);
console.log("re-granted portion   : " + grantPaths[1].join(", "));
console.log("stops.json fetches   : " + byteFetches.filter((u) => u === "/private/stops.json").length + " (verified file never refetched)");
console.log("audio fetch attempts : " + byteFetches.filter((u) => u.endsWith("audio/story-1.m4a")).length + " (1 expired + 1 fresh)");
console.log("final layer          : " + fs.readdirSync(root + "/bundles/route-x/1/be/extended").sort().join(", "));
' 2>/dev/null
```

```output
activation           : complete
grant rounds         : 2
re-granted portion   : audio/story-1.m4a
stops.json fetches   : 1 (verified file never refetched)
audio fetch attempts : 2 (1 expired + 1 fresh)
final layer          : audio, stops.json
```

Offline: the grant is unobtainable, the activation returns partial with a named reason and no exception reaches the caller; a plain retry completes (criterion 4):

```sh
node --experimental-strip-types -e 'const fs = await import("node:fs");
const { activate, parseLock } = await import("./services/download/download.ts");
const { createGrantFetchSource } = await import("./services/download/grant.ts");
const { createNodeDownloadStore, nodeSha256 } = await import("./services/download/nodeDownloadStore.ts");
const { openDatabase } = await import("./services/db/db.ts");
const { nodeSqliteDriver } = await import("./services/db/test-fixture.ts");
const root = "/tmp/g0402b-demo-3";
fs.rmSync(root, { recursive: true, force: true });
const enc = (s) => new TextEncoder().encode(s);
const stops = enc("{\"stops\":[]}\n");
const audio = enc("audio-bytes-0123456789abcdef");
const lock = [
  { path: "stops.json", bytes: stops.length, sha256: await nodeSha256(stops) },
  { path: "audio/story-1.m4a", bytes: audio.length, sha256: await nodeSha256(audio) },
];
const sources = { "stops.json": stops, "audio/story-1.m4a": audio };
const key = { routeId: "route-x", version: "1", locale: "be", tier: "extended" };
let online = false;
const transport = async (req) => {
  if (!online) throw new Error("network down");
  return { status: 200, headers: {}, body: { lock_url: "https://cdn.test/lock.json", urls: req.body.paths.map((p) => ({ path: p, url: "/private/" + p, expires_at: 600000 })) } };
};
const fetchBytes = async (url) => ({ status: 200, body: sources[url.slice("/private/".length)] });
const driver = nodeSqliteDriver();
openDatabase(driver);
const store = createNodeDownloadStore(root);
const makeFetch = () =>
  createGrantFetchSource(
    { key, entries: parseLock(lock).entries },
    { transport, credential: async () => "redacted", fetchBytes, delay: async () => {}, now: () => 0 },
  );
const offline = await activate({ ...key, lock }, { store, fetch: makeFetch(), sha256: nodeSha256, driver });
console.log("offline run   : status=" + offline.status + " (no exception reached the caller)");
console.log("  diagnostics : " + offline.diagnostics.join(" | "));
console.log("  final layer : " + (fs.existsSync(root + "/bundles/route-x/1/be/extended") ? "exists" : "absent"));
online = true;
const retry = await activate({ ...key, lock }, { store, fetch: makeFetch(), sha256: nodeSha256, driver });
console.log("retry run     : status=" + retry.status + ", fetched=" + retry.fetched);
console.log("  final layer : " + fs.readdirSync(root + "/bundles/route-x/1/be/extended").sort().join(", "));
' 2>/dev/null
```

  diagnostics : stops.json#fetch-failed | grant-fetch#grant-offline
  final layer : absent
retry run     : status=complete, fetched=2
  final layer : audio, stops.json
